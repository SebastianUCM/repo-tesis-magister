\subsection{sub sección2}\label{subsec1}

Ejemplo Algoritmo \ref{alg_ss}.

\begin{algorithm}[H]

    \begin{algorithmic}[1]
        \FOR{i = 0; i < tamaño - 1 ; i++}
                \STATE min\_id = i
            \FOR{j = i + 1; j < tamaño ; j++}
                \IF{Arr[j] < Arr[min\_id]}
                    \STATE min\_id = j
                \ENDIF
                
            \ENDFOR
                \IF{ min\_id != i}
                    \STATE Cambio(Arr[min\_id], Arr[i])
                    \ENDIF
         \ENDFOR
    
    \end{algorithmic}
    \caption{\label{alg_ss} Selection Sort(arr[ ], tamaño).}
\end{algorithm}
